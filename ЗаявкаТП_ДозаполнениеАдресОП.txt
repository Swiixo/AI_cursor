// =====================================================================================================
// ДОЗАПОЛНЕНИЕ АДРЕСА И ОБЪЕКТА ПРИСОЕДИНЕНИЯ В ЗАЯВКАХ ТП ИЗ КрымMySQL (модуль формы)
// =====================================================================================================
//
// ЗАЧЕМ
//   При исторической загрузке (ЗаявкаТП_ПоШагам) шаг 1 пропускал аннулированные заявки, и часть заявок
//   осталась без адреса подключения и/или объекта присоединения. Эта обработка находит такие заявки
//   и выполняет для них тот же шаг 1: ИБР_Адреса (кадастр из КрымPG), ЭЭ_ОбъектыПрисоединения,
//   ЭЭ_ЭнергопринимающиеУстройства, а в заявке — адрес, ОП, мощности по категориям, этап,
//   уровень напряжения, телефон, категория надёжности, вид присоединения.
//
// КАКИЕ ЗАЯВКИ ОБРАБАТЫВАЮТСЯ
//   Все ЭЭ_ЗаявкаТП из КрымMySQL (ЗаявкаПолученаССайта = БазаMySQL, ИБР_Основание = bid.BidID),
//   не помеченные на удаление, у которых пуст АдресПодключения ИЛИ пуст ОбъектПрисоединения —
//   аннулированные и любые другие (например, те, на которых шаг 1 раньше упал).
//   Заявки с пустым bid.Obj_Address пропускаются (без адреса не из чего создать ИБР_Адреса и ОП).
//   Заявки, BidID которых нет в bid, тоже пропускаются — их количество выводится в итогах.
//
// ПОВТОРНЫЙ ЗАПУСК
//   Безопасен: если у заявки уже есть адрес или ОП, они обновляются, а не создаются заново
//   (ЭПУ сопоставляются по категории надёжности и признаку «подключаемое»). Заявки, у которых
//   после запуска заполнены и адрес, и ОП, в следующий запуск не попадут.
//
// ОБЩЕЕ
//   * Объекты пишутся с ОбменДанными.Загрузка = Истина — обработчики записи и подписки не выполняются.
//   * Источники (ИБ, КрымMySQL, КрымPG) одним запросом не соединить: каждый читается отдельно
//     и сопоставляется в памяти по номеру заявки; к КрымMySQL — порциями по РазмерПорцииКрымMySQL().
//   * Каждая заявка пишется в своей транзакции; ошибка по одной заявке не останавливает обработку.
//     Ошибки и пропуски — в журнал «ИсторическиеЗаявки. Подготовка адреса».
//
// ЧТО НУЖНО СОЗДАТЬ НА ФОРМЕ
//   Реквизиты:
//     ПервыеN  - Число(10,0), неотрицательное - сколько заявок обработать (по возрастанию BidID): 0 = все.
//     Протокол - ТаблицаЗначений с колонкой ЗаявкаТП (произвольный тип) - заявки, отобранные в последний запуск.
//   Команды:
//     ДозаполнитьАдресаИОП - действие — одноимённая процедура &НаКлиенте.
//
// =====================================================================================================

// Контекст обработки: заполняется в ЗаполнитьАдресаЗаявок, читается в ОбработатьЗаявку.

	Сообщить("===== Дозаполнение адреса и ОП: начало " + Формат(ТекущаяДатаСеанса(), "ДЛФ=T"));
	Начало = ТекущаяУниверсальнаяДатаВМиллисекундах();

	Попытка
		Партия = ЗаявкиБезАдресаИлиОП(ПервыеN);
		ЗаполнитьПротокол(Партия);

		КонецЕсли;

	Исключение

		ЗаписьЖурналаРегистрации("ИсторическиеЗаявки. Дозаполнение адреса и ОП", УровеньЖурналаРегистрации.Ошибка,,,
			ПодробноеПредставлениеОшибки(ИнформацияОбОшибке()));
		Возврат;
	КонецПопытки;

	Секунд = Окр((ТекущаяУниверсальнаяДатаВМиллисекундах() - Начало) / 1000);
	Сообщить("===== Дозаполнение адреса и ОП: завершено за " + Формат(Секунд, "ЧН=0; ЧГ=0") + " сек.");

// Заявки КрымMySQL без адреса подключения или без объекта присоединения.
// Возвращает ТаблицаЗначений (BidID, Заявка) по возрастанию BidID; ПервыеN > 0 — только первые N.
&НаСервере
Функция ЗаявкиБезАдресаИлиОП(ПервыеN = 0)

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ
	|	ЭЭ_ЗаявкаТП.ИБР_Основание КАК BidID,
	|	МАКСИМУМ(ЭЭ_ЗаявкаТП.Ссылка) КАК Заявка
	|ИЗ
	|	Документ.ЭЭ_ЗаявкаТП КАК ЭЭ_ЗаявкаТП
	|ГДЕ
	|	ЭЭ_ЗаявкаТП.ЗаявкаПолученаССайта = ЗНАЧЕНИЕ(Справочник.УТП_ИсточникиЗаявок.БазаMySQL)
	|	И НЕ ЭЭ_ЗаявкаТП.ПометкаУдаления
	|	И ЭЭ_ЗаявкаТП.ИБР_Основание > 0
	|	И (ЭЭ_ЗаявкаТП.АдресПодключения = ЗНАЧЕНИЕ(Справочник.ИБР_Адреса.ПустаяСсылка)
	|			ИЛИ ЭЭ_ЗаявкаТП.АдресПодключения ЕСТЬ NULL
	|			ИЛИ ЭЭ_ЗаявкаТП.ОбъектПрисоединения = ЗНАЧЕНИЕ(Справочник.ЭЭ_ОбъектыПрисоединения.ПустаяСсылка)
	|			ИЛИ ЭЭ_ЗаявкаТП.ОбъектПрисоединения ЕСТЬ NULL)
	|
	|СГРУППИРОВАТЬ ПО
	|	ЭЭ_ЗаявкаТП.ИБР_Основание
	|
	|УПОРЯДОЧИТЬ ПО
	|	BidID";
	// ПЕРВЫЕ не принимает параметр — число подставляется в текст.
	Если ПервыеN > 0 Тогда
		Запрос.Текст = СтрЗаменить(Запрос.Текст, "ВЫБРАТЬ", "ВЫБРАТЬ ПЕРВЫЕ " + Формат(ПервыеN, "ЧГ=0"));
	КонецЕсли;
	Партия = Запрос.Выполнить().Выгрузить();

	Сообщить("Заявок без адреса или объекта присоединения" + ?(ПервыеN > 0, " (первые " + ПервыеN + ")", "")
		+ ": " + Партия.Количество());
	Возврат Партия;

// =====================================================================================================
#Область АдресаОбъектыПрисоединенияЭПУ
// =====================================================================================================
// КрымMySQL.bid -> ИБР_Адреса -> ЭЭ_ОбъектыПрисоединения
//               -> ЭЭ_ЭнергопринимающиеУстройства (по одному на каждую заполненную мощность)
//               -> ЭЭ_ЗаявкаТП: АдресПодключения, ОбъектПрисоединения, мощности по категориям, этап,
//                  УровеньНапряжения, Телефон (из bid.Obj_Phone, только если заполнен), вид присоединения.
// Кадастр берётся из КрымPG (privatecab_public_orders) по связи utp_number = BidID.
// Каждая заявка пишется в своей транзакции: адрес, ОП, ЭПУ и заявка — либо все, либо ничего.
//
// СОЗДАТЬ ИЛИ ОБНОВИТЬ:
//   * адрес — если в заявке уже есть АдресПодключения, обновляется он, иначе создаётся новый;
//   * объект присоединения — если в заявке уже есть ОбъектПрисоединения, обновляется он, иначе создаётся;
//   * ЭПУ — у существующего ОП ищутся его ЭПУ по паре «категория надёжности + подключаемое/ранее
//     присоединённое» (подключаемое = ЭПУ со ссылкой на заявку). Найдено — обновляется, нет — создаётся.
//     ЭПУ, для которых в bid больше нет мощности, помечаются на удаление.
&НаСервере
Процедура ЗаполнитьАдресаЗаявок(Партия)

	Заявки_BID_тз = ЗаявкиBID(Партия.ВыгрузитьКолонку("BidID"));
	Сообщить("Заявок к обработке: " + Партия.Количество() + ", найдено в bid: " + Заявки_BID_тз.Количество()
		+ ?(Партия.Количество() > Заявки_BID_тз.Количество(),
			" (нет в bid, пропущены: " + (Партия.Количество() - Заявки_BID_тз.Количество()) + ")", ""));

	// Разные базы (КрымMySQL, КрымPG, текущая ИБ) одним запросом не соединить,
	// поэтому каждая выгружается отдельно и сопоставляется в памяти по номеру заявки.
	Контекст = Новый Структура;
	// Номера партии строкой — в КрымPG номер заявки хранится текстом.
	НомераПартии = Новый Соответствие;
	Для Каждого СтрокаПартии Из Партия Цикл
		НомераПартии.Вставить(НомерСтрокой(СтрокаПартии.BidID), Истина);
	КонецЦикла;
	Контекст.Вставить("КадастрПоНомеру",  КадастрыЗаявок(НомераПартии));
	Контекст.Вставить("ЗаявкиПоBidID",    ЗаявкиПоBidID(Партия));
	Контекст.Вставить("УровниНапряжения", УровниНапряжения());
	Контекст.Вставить("Обработано", 0);
	Контекст.Вставить("Обновлено",  0); // из обработанных: у заявки уже были адрес или ОП
	Контекст.Вставить("Пропущено",  0);
	Контекст.Вставить("Ошибок",     0);

	Для Каждого СтрокаBID Из Заявки_BID_тз Цикл
		ОбработатьЗаявку(СтрокаBID);
	КонецЦикла;

	Сообщить("Обработано: " + Контекст.Обработано + " (из них обновлено существующих: " + Контекст.Обновлено + ")"
		+ ", пропущено: " + Контекст.Пропущено
		+ ", ошибок: " + Контекст.Ошибок + " (подробности — журнал ""ИсторическиеЗаявки. Подготовка адреса"")");

КонецПроцедуры

&НаСервере
Процедура ОбработатьЗаявку(СтрокаBID)

	НомерЗаявки = НомерСтрокой(СтрокаBID.BidID);
	Адрес       = СтрокаИлиПусто(СтрокаBID.Obj_Address);
	Кадастр     = Контекст.КадастрПоНомеру[НомерЗаявки];
	Заявка      = Контекст.ЗаявкиПоBidID[СтрокаBID.BidID];

	// Без адреса не из чего создавать ИБР_Адреса, а без него — объект присоединения.
	Если Не ЗначениеЗаполнено(Адрес) Тогда
		ЗаписатьОшибку(НомерЗаявки, Заявка, "Obj_Address пустой - пропуск");
		Контекст.Пропущено = Контекст.Пропущено + 1;
		Возврат;
	КонецЕсли;

	// --- данные объекта присоединения ---

	НаименованиеОП    = СтрокаИлиПусто(СтрокаBID.Obj_Name);
	НапряжениеBID     = СтрокаИлиПусто(СтрокаBID.Bid_Voltage);
	ТелефонBID        = СтрокаИлиПусто(СтрокаBID.Obj_Phone);
	УровеньНапряжения = УровеньНапряженияПоЗначению(НапряжениеBID, Контекст.УровниНапряжения);

	Если ЗначениеЗаполнено(НапряжениеBID) И Не ЗначениеЗаполнено(УровеньНапряжения) Тогда
		ЗаписатьОшибку(НомерЗаявки, Заявка, "Значение """ + НапряжениеBID + """ нет в соответствии - уровень напряжения пустой",
			УровеньЖурналаРегистрации.Предупреждение);
	КонецЕсли;

	// Мощности для объекта присоединения: полная (Sum) и ранее присоединённая (Cur) по категориям.
	Мощности = Новый Структура;
	Мощности.Вставить("Категория1",      ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1));
	Мощности.Вставить("Категория2",      ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2));
	Мощности.Вставить("Категория3",      ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3));
	Мощности.Вставить("РанееКатегория1", ЧислоИлиНоль(СтрокаBID.Bid_PowerCur1));
	Мощности.Вставить("РанееКатегория2", ЧислоИлиНоль(СтрокаBID.Bid_PowerCur2));
	Мощности.Вставить("РанееКатегория3", ЧислоИлиНоль(СтрокаBID.Bid_PowerCur3));

	// Мощности для заявки: полная по категориям и прирост (Sum - Cur) по категориям.
	Мощности_заяв = Новый Структура;
	Мощности_заяв.Вставить("ПолнаяКатегория1", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1));
	Мощности_заяв.Вставить("ПолнаяКатегория2", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2));
	Мощности_заяв.Вставить("ПолнаяКатегория3", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3));
	Мощности_заяв.Вставить("Категория1", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur1));
	Мощности_заяв.Вставить("Категория2", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur2));
	Мощности_заяв.Вставить("Категория3", ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur3));

	ОписанияЭПУ_Мс = ОписанияЭПУ(СтрокаBID, НаименованиеОП);

	// --- запись: адрес -> ОП -> ЭПУ -> заявка, одной транзакцией ---

	НачатьТранзакцию();
	Попытка

		// Заявка читается первой: в ней ссылки на адрес и ОП, если они уже были созданы.
		ДокОбъект = Заявка.ПолучитьОбъект();
		ЕстьАдрес = ЗначениеЗаполнено(ДокОбъект.АдресПодключения);
		ЕстьОП    = ЗначениеЗаполнено(ДокОбъект.ОбъектПрисоединения);

		// Адрес: есть в заявке — обновляем, нет — создаём.
		Если ЕстьАдрес Тогда
			АдресОбъект = ДокОбъект.АдресПодключения.ПолучитьОбъект();
		Иначе
			АдресОбъект = Справочники.ИБР_Адреса.СоздатьЭлемент();
		КонецЕсли;
		АдресОбъект.Наименование            = Адрес;
		АдресОбъект.АдресПредставление      = Адрес;
		АдресОбъект.КадастровыйНомерУчастка = ?(Кадастр = Неопределено, "", Кадастр);
		АдресОбъект.ИБР_Основание           = СтрокаBID.BidID;
		АдресОбъект.ОбменДанными.Загрузка = Истина;
		АдресОбъект.Записать();

		// Объект присоединения (ссылается на адрес): есть в заявке — обновляем, нет — создаём.
		Если ЕстьОП Тогда
			ОПОбъект = ДокОбъект.ОбъектПрисоединения.ПолучитьОбъект();
		Иначе
			ОПОбъект = Справочники.ЭЭ_ОбъектыПрисоединения.СоздатьЭлемент();
		КонецЕсли;
		ОПОбъект.Наименование = НаименованиеОП;
		ОПОбъект.Адрес        = АдресОбъект.Ссылка;
		ЗаполнитьЗначенияСвойств(ОПОбъект, Мощности);
		ОПОбъект.РанееРазрешеннаяМощность = ОПОбъект.РанееКатегория1 + ОПОбъект.РанееКатегория2 + ОПОбъект.РанееКатегория3;
		ОПОбъект.УровеньНапряжения = УровеньНапряжения;
		ОПОбъект.ИБР_Основание     = СтрокаBID.BidID;
		ОПОбъект.ОбменДанными.Загрузка = Истина;
		ОПОбъект.Записать();

		// ЭПУ (владелец — объект присоединения; подключаемые ЭПУ ссылаются на заявку).
		// У нового ОП ЭПУ ещё нет; у существующего — берём его ЭПУ и сопоставляем по ключу.
		СуществующиеЭПУ = ?(ЕстьОП, НайтиСуществующиеЭПУ(ОПОбъект.Ссылка), Новый Соответствие);
		Для Каждого ЭПУ Из ОписанияЭПУ_Мс Цикл
			КлючЭПУ = СформироватьКлючЭПУ(ЭПУ.Подключаемая, ЭПУ.Категория);
			СсылкаЭПУ = СуществующиеЭПУ[КлючЭПУ];
			Если СсылкаЭПУ <> Неопределено Тогда
				ЭПУОбъект = СсылкаЭПУ.ПолучитьОбъект();
				СуществующиеЭПУ.Удалить(КлючЭПУ); // сопоставлено — не удалять ниже
			Иначе
				ЭПУОбъект = Справочники.ЭЭ_ЭнергопринимающиеУстройства.СоздатьЭлемент();
			КонецЕсли;
			ЭПУОбъект.Владелец            = ОПОбъект.Ссылка;
			ЭПУОбъект.Наименование        = ЭПУ.Наименование;
			ЭПУОбъект.Мощность            = ЭПУ.Мощность;
			ЭПУОбъект.КатегорияНадежности = ЭПУ.Категория;
			ЭПУОбъект.УровеньНапряжения   = УровеньНапряжения;
			Если ЭПУ.Подключаемая Тогда
				ЭПУОбъект.Заявка = Заявка;
			КонецЕсли;
			ЭПУОбъект.ИБР_Основание = СтрокаBID.BidID;
			ЭПУОбъект.ОбменДанными.Загрузка = Истина;
			ЭПУОбъект.Записать();
		КонецЦикла;

		// Оставшиеся ЭПУ существующего ОП в bid больше не встречаются (мощность стала 0) — на удаление.
		Для Каждого КлючИЗначение Из СуществующиеЭПУ Цикл
			ЭПУОбъект = КлючИЗначение.Значение.ПолучитьОбъект();
			ЭПУОбъект.ПометкаУдаления = Истина;
			ЭПУОбъект.ОбменДанными.Загрузка = Истина;
			ЭПУОбъект.Записать();
		КонецЦикла;

		// Заявка (объект получен в начале транзакции). Этапы пересобираются заново.
		ДокОбъект.Этапы.Очистить();

		ДокОбъект.АдресПодключения    = АдресОбъект.Ссылка;
		ДокОбъект.ОбъектПрисоединения = ОПОбъект.Ссылка;
		ЗаполнитьЗначенияСвойств(ДокОбъект, Мощности_заяв);
		ДокОбъект.ЕдиновременнаяНагрузка = Мощности_заяв.ПолнаяКатегория1 + Мощности_заяв.ПолнаяКатегория2 + Мощности_заяв.ПолнаяКатегория3;

		ДокОбъект.УровеньНапряжения = УровеньНапряжения;
		Если ЗначениеЗаполнено(ТелефонBID) Тогда
			ДокОбъект.Телефон = ТелефонBID;
		КонецЕсли;

		ДокОбъект.МощностьПоКатегориямНадежности = ""
			+ ?(ДокОбъект.Категория1 > 0, "первая (" + ДокОбъект.Категория1 + "кВт), ", "")
			+ ?(ДокОбъект.Категория2 > 0, "вторая (" + ДокОбъект.Категория2 + "кВт), ", "")
			+ ?(ДокОбъект.Категория3 > 0, "третья (" + ДокОбъект.Категория3 + "кВт) ", "");

		// Один этап «Полное подключение» на всю мощность (этапы очищены выше, итог = 0).
		МощностьЭтапа = ДокОбъект.ЕдиновременнаяНагрузка - ДокОбъект.Этапы.Итог("Мощность");
		Если МощностьЭтапа > 0 Тогда
			НЭ = ДокОбъект.Этапы.Добавить();
			НЭ.НазваниеЭтапа = "Полное подключение";
			НЭ.Мощность   = МощностьЭтапа;
			НЭ.Категория1 = ДокОбъект.Категория1 - ДокОбъект.Этапы.Итог("Категория1");
			НЭ.Категория2 = ДокОбъект.Категория2 - ДокОбъект.Этапы.Итог("Категория2");
			НЭ.Категория3 = ДокОбъект.Категория3 - ДокОбъект.Этапы.Итог("Категория3");
		КонецЕсли;

		// Категория надёжности заявки — самая высокая из заполненных.
		ДокОбъект.КатегорияНадежности = ?(ДокОбъект.ПолнаяКатегория1 > 0, Справочники.ЭЭ_КатегорииНадежности.Категория1,
			?(ДокОбъект.ПолнаяКатегория2 > 0, Справочники.ЭЭ_КатегорииНадежности.Категория2, Справочники.ЭЭ_КатегорииНадежности.Категория3));
		ДокОбъект.ПолнаяМощность = ДокОбъект.ОбъектПрисоединения.РанееРазрешеннаяМощность + ДокОбъект.ЕдиновременнаяНагрузка;

		Для Каждого ТекЭтап Из ДокОбъект.Этапы Цикл
			ТекЭтап.СписокКатегорий = "" + ?(ТекЭтап.Категория1 > 0, "1,", "") + ?(ТекЭтап.Категория2 > 0, " 2,", "") + ?(ТекЭтап.Категория3 > 0, " 3", "");
		КонецЦикла;

		// Вид присоединения — окончательно, по объекту присоединения (на шаге 0 был предварительный).
		ДокОбъект.НаУвеличениеМощности   = (ДокОбъект.ОбъектПрисоединения.РанееРазрешеннаяМощность > 0);
		ДокОбъект.ВидообразующийРеквизит = ?(ДокОбъект.НаУвеличениеМощности,
			Справочники.ИБР_ВидыОбъектов.Увеличение, Справочники.ИБР_ВидыОбъектов.ПервичноеПрисоединение);

		ДокОбъект.ОбменДанными.Загрузка = Истина;
		ДокОбъект.Записать();

		ЗафиксироватьТранзакцию();
		Контекст.Обработано = Контекст.Обработано + 1;
		Если ЕстьАдрес ИЛИ ЕстьОП Тогда
			Контекст.Обновлено = Контекст.Обновлено + 1;
		КонецЕсли;

	Исключение

		ОтменитьТранзакцию();
		ЗаписатьОшибку(НомерЗаявки, Заявка, ПодробноеПредставлениеОшибки(ИнформацияОбОшибке()));
		Контекст.Ошибок = Контекст.Ошибок + 1;

	КонецПопытки;

// во временную таблицу ИБ: выборка проходится один раз, свёртка «последняя дата по номеру» делается в памяти,

// Запрос к КрымMySQL по списку BidID любой длины. Метка &СписокBidID в тексте заменяется списком чисел
// (параметр-массив 1С передаёт в MySQL через временную таблицу, а права CREATE TEMPORARY TABLES у onec нет).
// Список режется на порции по РазмерПорцииКрымMySQL(); результаты порций складываются в одну таблицу.
&НаСервере
Функция ВыполнитьЗапросКрымMySQLПорциями(ТекстЗапроса, СписокBidID)

	РазмерПорции = РазмерПорцииКрымMySQL();
	Результат = Неопределено;
	Начало = 0;
	Пока Начало < СписокBidID.Количество() Цикл

		Порция = Новый Массив;
		Для Индекс = Начало По Мин(Начало + РазмерПорции, СписокBidID.Количество()) - 1 Цикл
			Порция.Добавить(Формат(СписокBidID[Индекс], "ЧГ=0; ЧН=0"));
		КонецЦикла;
		Начало = Начало + РазмерПорции;

		Запрос = Новый Запрос(СтрЗаменить(ТекстЗапроса, "&СписокBidID", СтрСоединить(Порция, ", ")));
		ТаблицаПорции = Запрос.Выполнить().Выгрузить();

		Если Результат = Неопределено Тогда
			Результат = ТаблицаПорции;
		Иначе
			Для Каждого СтрокаПорции Из ТаблицаПорции Цикл
				ЗаполнитьЗначенияСвойств(Результат.Добавить(), СтрокаПорции);
			КонецЦикла;
		КонецЕсли;

	КонецЦикла;

	Возврат Результат;

КонецФункции

// Соответствие BidID -> ссылка на заявку партии.
