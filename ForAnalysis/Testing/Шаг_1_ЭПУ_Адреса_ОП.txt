Перем Контекст;

// Историческая перегрузка. Этап 1: адреса, объекты присоединения, ЭПУ.
// КрымMySQL.bid -> новый элемент ИБР_Адреса -> новый элемент ЭЭ_ОбъектыПрисоединения
//               -> ЭЭ_ЭнергопринимающиеУстройства (по одному на каждую заполненную мощность)
//               -> ЭЭ_ЗаявкаТП.АдресПодключения, ЭЭ_ЗаявкаТП.ОбъектПрисоединения,
//                  ЭЭ_ЗаявкаТП.ЭнергопринимающиеУстройства (только подключаемые ЭПУ),
//                  ЭЭ_ЗаявкаТП.Телефон (из bid.Obj_Phone, только если заполнен)
// Документ ищется по ИБР_Основание = BidID (оба Число) и ЗаявкаПолученаССайта = УТП_ИсточникиЗаявок.БазаMySQL.
// Кадастр берётся из КрымPG (privatecab_public_orders) по связи utp_number = BidID = НомерЗаявки.
// Аннулированные заявки (в bid заполнен Annul_NN или Annul_Date) не обрабатываются.
// Ошибки и пропуски пишутся в журнал регистрации "ИсторическиеЗаявки. Подготовка адреса".
// Все объекты (справочники и документ) пишутся с ОбменДанными.Загрузка = Истина - без обработчиков записи.

// Параметры:
//  КоличествоЗаявок  - Число - сколько заявок bid обработать (первые неаннулированные по дате заявки).
//                      Указано (> 0) - тестовый прогон: в базу ничего не пишется,
//                      в ТаблицаРезультата попадает то, что было бы сделано.
//                      0 / Неопределено - боевой прогон по всем заявкам bid, протокол не ведётся.
//  ТаблицаРезультата - ТаблицаЗначений, Неопределено - (возвращаемый) протокол тестового прогона.
//
Процедура ПерегрузитьАдресаЗаявок(КоличествоЗаявок = 0, ТаблицаРезультата = Неопределено) Экспорт

	ТестовыйПрогон = Ложь;//ЗначениеЗаполнено(КоличествоЗаявок);
	ТаблицаРезультата = ?(Истина, НоваяТаблицаРезультата(), Неопределено);

	Заявки_BID_тз = ЗаявкиBID(КоличествоЗаявок);
	Если Заявки_BID_тз.Количество() = 0 Тогда
		Возврат;
	КонецЕсли;

	// Разные базы (КрымMySQL, КрымPG, текущая ИБ) - одним запросом не соединить,
	// поэтому каждая выгружается отдельно и сопоставляется в памяти по номеру заявки.
	Контекст = Новый Структура;
	Контекст.Вставить("ТестовыйПрогон",    ТестовыйПрогон);
	Контекст.Вставить("ТаблицаРезультата", ТаблицаРезультата);
	Контекст.Вставить("КадастрПоНомеру",   КадастрыЗаявок());
	Контекст.Вставить("ЗаявкеТППоНомеру", ДокументыЗаявокТП(Заявки_BID_тз));
	Контекст.Вставить("УровниНапряжения",  УровниНапряжения());
	Контекст.Вставить("ОснованиеАдресаЧисло", Истина);
	Контекст.Вставить("ОснованиеОПЧисло",     Истина);
	Контекст.Вставить("ОснованиеЭПУЧисло",    Истина);
	//СчетчикЗаяки = 0;
	Для Каждого СтрокаBID Из Заявки_BID_тз Цикл
		//Если СчетчикЗаяки > КоличествоЗаявок Тогда
		//	Прервать;
		//КонецЕсли;
		
		ОбработатьЗаявку(СтрокаBID);
		//СчетчикЗаяки = СчетчикЗаяки + 1;
	КонецЦикла;

КонецПроцедуры

Процедура ОбработатьЗаявку(СтрокаBID)

	ТаблицаРезультата = Контекст.ТаблицаРезультата;

	НомерЗаявки        = НомерСтрокой(СтрокаBID.BidID);
	Адрес              = СтрокаИлиПусто(СтрокаBID.Obj_Address);
	Кадастр            = Контекст.КадастрПоНомеру[НомерЗаявки];
	НайденныеДокументы = Контекст.ЗаявкеТППоНомеру[НомерЗаявки];

	// --- проверки ---

	Если НайденныеДокументы = Неопределено Тогда
		ТекстОшибки = "Документ с ИБР_Основание = BidID и источником БазаMySQL не найден - пропуск";
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ЗаявкаТП", "КрымMySQL.bid.BidID", "ЭЭ_ЗаявкаТП.ИБР_Основание",
			НомерЗаявки, "", ТекстОшибки);
		ЗаписатьОшибку(НомерЗаявки, Неопределено, ТекстОшибки);
		Возврат;
	ИначеЕсли НайденныеДокументы.Количество() > 1 Тогда
		ТекстОшибки = "Найдено документов: " + НайденныеДокументы.Количество() + " - пропуск";
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ЗаявкаТП", "КрымMySQL.bid.BidID", "ЭЭ_ЗаявкаТП.ИБР_Основание",
			НомерЗаявки, "", ТекстОшибки);
		ЗаписатьОшибку(НомерЗаявки, НайденныеДокументы[0].Ссылка, ТекстОшибки);
		Возврат;
	КонецЕсли;

	ДанныеДокумента_ЗаявкаТП_сс = НайденныеДокументы[0];

	Если Не ЗначениеЗаполнено(Адрес) Тогда
		ТекстОшибки = "Obj_Address пустой - пропуск";
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, "КрымMySQL.bid.Obj_Address", "ИБР_Адреса",
			"", "", ТекстОшибки);
		ЗаписатьОшибку(НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, ТекстОшибки);
		Возврат;
	КонецЕсли;

	// --- данные объекта присоединения ---

	НаименованиеОП = СтрокаИлиПусто(СтрокаBID.Obj_Name);
	НапряжениеBID  = СтрокаИлиПусто(СтрокаBID.Bid_Voltage);
	ТелефонBID     = СтрокаИлиПусто(СтрокаBID.Obj_Phone);
	УровеньНапряжения = УровеньНапряженияПоЗначению(НапряжениеBID, Контекст.УровниНапряжения);

	КомментарийНапряжения = "";
	Если ЗначениеЗаполнено(НапряжениеBID) И Не ЗначениеЗаполнено(УровеньНапряжения) Тогда
		КомментарийНапряжения = "Значение """ + НапряжениеBID + """ нет в соответствии - уровень напряжения пустой";
		ЗаписатьОшибку(НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, КомментарийНапряжения, УровеньЖурналаРегистрации.Предупреждение);
	КонецЕсли;

	Мощности = Новый Структура;
	Мощности.Вставить("Категория1",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1));
	Мощности.Вставить("Категория2",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2));
	Мощности.Вставить("Категория3",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3));
	Мощности.Вставить("РанееКатегория1",  ЧислоИлиНоль(СтрокаBID.Bid_PowerCur1));
	Мощности.Вставить("РанееКатегория2",  ЧислоИлиНоль(СтрокаBID.Bid_PowerCur2));
	Мощности.Вставить("РанееКатегория3",  ЧислоИлиНоль(СтрокаBID.Bid_PowerCur3));
	
	Мощности_заяв = Новый Структура;
	Мощности_заяв.Вставить("ПолнаяКатегория1",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1));
	Мощности_заяв.Вставить("ПолнаяКатегория2",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2));
	Мощности_заяв.Вставить("ПолнаяКатегория3",       ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3));
	Мощности_заяв.Вставить("Категория1",  ЧислоИлиНоль(СтрокаBID.Bid_PowerSum1) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur1));
	Мощности_заяв.Вставить("Категория2",  ЧислоИлиНоль(СтрокаBID.Bid_PowerSum2) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur2));
	Мощности_заяв.Вставить("Категория3",  ЧислоИлиНоль(СтрокаBID.Bid_PowerSum3) - ЧислоИлиНоль(СтрокаBID.Bid_PowerCur3));
	
	ИсточникМощности = Новый Структура;
	ИсточникМощности.Вставить("Категория1",      "Bid_PowerSum1");
	ИсточникМощности.Вставить("Категория2",      "Bid_PowerSum2");
	ИсточникМощности.Вставить("Категория3",      "Bid_PowerSum3");
	ИсточникМощности.Вставить("РанееКатегория1", "Bid_PowerCur1");
	ИсточникМощности.Вставить("РанееКатегория2", "Bid_PowerCur2");
	ИсточникМощности.Вставить("РанееКатегория3", "Bid_PowerCur3");

	ОписанияЭПУ_Мс = ОписанияЭПУ(СтрокаBID, НаименованиеОП);

	// --- тестовый прогон: только протокол, что куда пойдёт ---

	Если Контекст.ТестовыйПрогон Тогда

		ИсточникАдреса = "КрымMySQL.bid.Obj_Address";
		НовыйЭлемент   = "<новый элемент>";

		// Адрес
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ИБР_Адреса", ИсточникАдреса, "ИБР_Адреса.Наименование",
			Адрес, НовыйЭлемент, КомментарийДлины(Адрес, Метаданные.Справочники.ИБР_Адреса.ДлинаНаименования));
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ИБР_Адреса", ИсточникАдреса, "ИБР_Адреса.АдресПредставление",
			Адрес, НовыйЭлемент, КомментарийДлины(Адрес, ДлинаРеквизита(Метаданные.Справочники.ИБР_Адреса, "АдресПредставление")));
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ИБР_Адреса", "КрымPG.privatecab_public_orders (utp_number = BidID)",
			"ИБР_Адреса.КадастровыйНомерУчастка", Кадастр, НовыйЭлемент,
			?(ЗначениеЗаполнено(Кадастр), "", "Кадастр не найден - реквизит останется пустым"));
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ИБР_Адреса", "КрымMySQL.bid.BidID", "ИБР_Адреса.ИБР_Основание",
			СтрокаBID.BidID, НовыйЭлемент, "");

		// Объект присоединения
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ОбъектыПрисоединения", "ИБР_Адреса (новый)",
			"ЭЭ_ОбъектыПрисоединения.Адрес", Адрес, НовыйЭлемент, "");
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ОбъектыПрисоединения", "КрымMySQL.bid.Obj_Name",
			"ЭЭ_ОбъектыПрисоединения.Наименование", НаименованиеОП, НовыйЭлемент,
			КомментарийДлины(НаименованиеОП, Метаданные.Справочники.ЭЭ_ОбъектыПрисоединения.ДлинаНаименования));
		Для Каждого КлючЗначение Из Мощности Цикл
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ОбъектыПрисоединения",
				"КрымMySQL.bid." + ИсточникМощности[КлючЗначение.Ключ],
				"ЭЭ_ОбъектыПрисоединения." + КлючЗначение.Ключ, КлючЗначение.Значение, НовыйЭлемент, "");
		КонецЦикла;
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ОбъектыПрисоединения", "КрымMySQL.bid.Bid_Voltage",
			"ЭЭ_ОбъектыПрисоединения.УровеньНапряжения", НапряжениеBID, УровеньНапряжения, КомментарийНапряжения);
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ОбъектыПрисоединения", "КрымMySQL.bid.BidID",
			"ЭЭ_ОбъектыПрисоединения.ИБР_Основание", СтрокаBID.BidID, НовыйЭлемент, "");

		// ЭПУ
		Если ОписанияЭПУ_Мс.Количество() = 0 Тогда
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, "ЭЭ_ЭнергопринимающиеУстройства", "КрымMySQL.bid.Bid_Power*",
				"ЭЭ_ЭнергопринимающиеУстройства", "", "", "Нет заполненных мощностей - ЭПУ не создаются");
		КонецЕсли;
		
		Для Каждого ЭПУ Из ОписанияЭПУ_Мс Цикл
			ОбъектЭПУ = "ЭЭ_ЭнергопринимающиеУстройства (" + ЭПУ.ПолеBID + ")";
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "КрымMySQL.bid." + ЭПУ.ПолеBID,
				"ЭЭ_ЭнергопринимающиеУстройства.Мощность", ЭПУ.Мощность, НовыйЭлемент, "");
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "КрымMySQL.bid." + ЭПУ.ПолеBID + " (цифра в имени поля)",
				"ЭЭ_ЭнергопринимающиеУстройства.КатегорияНадежности", ЭПУ.ПолеBID, ЭПУ.Категория, "");
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "КрымMySQL.bid.Bid_Voltage",
				"ЭЭ_ЭнергопринимающиеУстройства.УровеньНапряжения", НапряжениеBID, УровеньНапряжения, КомментарийНапряжения);
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "ЭЭ_ОбъектыПрисоединения (новый)",
				"ЭЭ_ЭнергопринимающиеУстройства.Владелец", НаименованиеОП, НовыйЭлемент, "");
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "ЭЭ_ЗаявкаТП",
				"ЭЭ_ЭнергопринимающиеУстройства.Заявка", ?(ЭПУ.Подключаемая, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, ""), НовыйЭлемент,
				?(ЭПУ.Подключаемая, "", "Ранее присоединённая мощность - заявка не указывается"));
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "КрымMySQL.bid.BidID",
				"ЭЭ_ЭнергопринимающиеУстройства.ИБР_Основание", СтрокаBID.BidID, НовыйЭлемент, "");
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ОбъектЭПУ, "Подкл./Ранее + категория + КрымMySQL.bid.Obj_Name",
				"ЭЭ_ЭнергопринимающиеУстройства.Наименование", ЭПУ.Наименование, НовыйЭлемент,
				КомментарийДлины(ЭПУ.Наименование, Метаданные.Справочники.ЭЭ_ЭнергопринимающиеУстройства.ДлинаНаименования));
		КонецЦикла;

		// Документ
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, "ИБР_Адреса (новый)", "ЭЭ_ЗаявкаТП.АдресПодключения",
			Адрес, ДанныеДокумента_ЗаявкаТП_сс.АдресПодключения,
			?(ЗначениеЗаполнено(ДанныеДокумента_ЗаявкаТП_сс.АдресПодключения), "Текущий адрес будет заменён", ""));
		ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, "ЭЭ_ОбъектыПрисоединения (новый)",
			"ЭЭ_ЗаявкаТП.ОбъектПрисоединения", НаименованиеОП, ДанныеДокумента_ЗаявкаТП_сс.ОбъектПрисоединения,
			?(ЗначениеЗаполнено(ДанныеДокумента_ЗаявкаТП_сс.ОбъектПрисоединения), "Текущий объект присоединения будет заменён", ""));
		Если ЗначениеЗаполнено(ТелефонBID) Тогда
			ДобавитьСтрокуВТаблицуПротокола(ТаблицаРезультата, НомерЗаявки, ДанныеДокумента_ЗаявкаТП_сс.Ссылка, "КрымMySQL.bid.Obj_Phone",
				"ЭЭ_ЗаявкаТП.Телефон", ТелефонBID, ДанныеДокумента_ЗаявкаТП_сс.Телефон,
					"Текущий телефон будет заменён");
		КонецЕсли;

		Возврат;

	КонецЕсли;

	// --- боевой прогон: запись ---

	НачатьТранзакцию();
	Попытка

		АдресОбъект = Справочники.ИБР_Адреса.СоздатьЭлемент();
		АдресОбъект.Наименование            = Адрес;
		АдресОбъект.АдресПредставление      = Адрес;
		АдресОбъект.КадастровыйНомерУчастка = ?(Кадастр = Неопределено, "", Кадастр);
		АдресОбъект.ИБР_Основание           = ?(Контекст.ОснованиеАдресаЧисло, СтрокаBID.BidID, НомерЗаявки);
		АдресОбъект.ОбменДанными.Загрузка = Истина;
		АдресОбъект.Записать();

		ОПОбъект = Справочники.ЭЭ_ОбъектыПрисоединения.СоздатьЭлемент();
		ОПОбъект.Наименование      = НаименованиеОП;
		ОПОбъект.Адрес             = АдресОбъект.Ссылка;
		ЗаполнитьЗначенияСвойств(ОПОбъект, Мощности);
		
		ОПОбъект.РанееРазрешеннаяМощность = ОПОбъект.РанееКатегория1 + ОПОбъект.РанееКатегория2 + ОПОбъект.РанееКатегория3;
		ОПОбъект.УровеньНапряжения = УровеньНапряжения;

			КонецЕсли;
			ЭПУОбъект.ИБР_Основание       = ?(Контекст.ОснованиеЭПУЧисло, СтрокаBID.BidID, НомерЗаявки);

		//	ДокОбъект.Ссылка, ДокОбъект.Ссылка, "");

&НаСервере
Процедура ЗапуститьОбработкуНаСервере()
	
	Таблица_ = Неопределено;
	ПерегрузитьАдресаЗаявок(КоличествоЗаявок, Таблица_);
	
	ВеткиДерева = ТаблицаЗнач.ПолучитьЭлементы();
	ВеткиДерева.Очистить();
	
	Если Таблица_ = Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	УзлыЗаявок = Новый Соответствие; // BidID -> строка заявки
	Группы     = Новый Соответствие; // ключ группы -> строка группы
	Изменений  = Новый Соответствие; // BidID -> количество изменений
	
	ИмяЭПУ = "ЭЭ_ЭнергопринимающиеУстройства";
	
	Для Каждого ТС Из Таблица_ Цикл
		
		// --- уровень 1: заявка ---
		Узел = УзлыЗаявок[ТС.BidID];
		Если Узел = Неопределено Тогда
			Узел = ВеткиДерева.Добавить();
			Узел.BidID           = ТС.BidID;
			Узел.ОбъектИзменения = "ЭЭ_ЗаявкаТП";
			УзлыЗаявок.Вставить(ТС.BidID, Узел);
			Изменений.Вставить(ТС.BidID, 0);
		КонецЕсли;
		
		Если ТипЗнч(ТС.ОбъектИзменения) = Тип("ДокументСсылка.ЭЭ_ЗаявкаТП") Тогда
			Узел.ОбъектИзменения = ТС.ОбъектИзменения;
		КонецЕсли;
		
		// пропущенная заявка - одна строка с причиной, показываем её прямо на узле
		Если СтрНайти(ТС.Комментарий, "пропуск") > 0 Тогда
			ЗаполнитьЗначенияСвойств(Узел, ТС);
			Продолжить;
		КонецЕсли;
		
				// --- уровень 2-3: группа объекта ---
		ИмяОбъекта = Строка(ТС.ОбъектИзменения);
		
		Если ИмяОбъекта = "ИБР_Адреса" Тогда
			Родитель = ГруппаДерева(Узел, Группы, ТС.BidID + "|Адрес", "Адрес");
			
		ИначеЕсли ИмяОбъекта = "ЭЭ_ОбъектыПрисоединения" Тогда
			Родитель = ГруппаДерева(Узел, Группы, ТС.BidID + "|ОП", "Объект присоединения");
			
		ИначеЕсли СтрНачинаетсяС(ИмяОбъекта, ИмяЭПУ) Тогда
			Родитель = ГруппаДерева(Узел, Группы, ТС.BidID + "|ЭПУ", "ЭПУ");
			Если ИмяОбъекта <> ИмяЭПУ Тогда // конкретное ЭПУ, а не строка "нет заполненных мощностей"
				ПолеBID = СтрЗаменить(СтрЗаменить(ИмяОбъекта, ИмяЭПУ + " (", ""), ")", "");
				НомерЭПУ = Родитель.ПолучитьЭлементы().Количество() + 1;
				Родитель = ГруппаДерева(Родитель, Группы, ТС.BidID + "|ЭПУ|" + ПолеBID,
					"ЭПУ" + НомерЭПУ + " (" + ПолеBID + ")");
			КонецЕсли;
			
		Иначе
			// реквизиты самой заявки (адрес, ОП, строки ТЧ) - прямо под заявкой
			Родитель = Узел;
		КонецЕсли;
		
		Дочерняя = Родитель.ПолучитьЭлементы().Добавить();
		ЗаполнитьЗначенияСвойств(Дочерняя, ТС, , "BidID");
		
		Изменений[ТС.BidID] = Изменений[ТС.BidID] + 1;
		
	КонецЦикла;
	
	// на узле заявки - сколько изменений
	Для Каждого Узел Из ВеткиДерева Цикл
		Если Изменений[Узел.BidID] > 0 Тогда
			Узел.Комментарий = "Изменений: " + Изменений[Узел.BidID];
		КонецЕсли;
	КонецЦикла;
	
КонецПроцедуры


// Находит или создаёт строку-группу под Родитель. НомерЭПУ считается только при создании группы.
&НаСервере
Функция ГруппаДерева(Родитель, Группы, Ключ, Имя)
	
	Группа = Группы[Ключ];
	Если Группа = Неопределено Тогда
		Группа = Родитель.ПолучитьЭлементы().Добавить();
		Группа.ОбъектИзменения = Имя;
		Группы.Вставить(Ключ, Группа);
	КонецЕсли;
	
	Возврат Группа;

	ЗапуститьОбработкуНаСервере();
