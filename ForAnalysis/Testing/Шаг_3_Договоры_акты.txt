	Команда1НаСервере();

&НаСервере
Процедура Команда1НаСервере()
	Валюта_ = Справочники.Валюты.НайтиПоКоду("643");
	ОснКомпания = Справочники.Контрагенты.ОсновнаяКомпания;
	ЛьготнаяКатегория_ = Справочники.УТП_КатегорииЛьгот.ЛьготнаяКатегория;
	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	bid.Ссылка КАК Ссылка,
		|	bid.Dogovor_NN КАК Dogovor_NN,
		|	bid.Dogovor_Date КАК Dogovor_Date,
		|	bid.SocTarif КАК SocTarif,
		|	cot.Ссылка КАК Ссылка1,
		|	cot.CotID КАК CotID,
		|	bid.BidID КАК BidID,
		|	cot.RealAct_NN КАК RealAct_NN,
		|	cot.RealAct_Date КАК RealAct_Date,
		|	cot.CostTotal КАК CostTotal,
		|	bid.EvtSrok КАК EvtSrok,
		|	evt.Vyp_k_Sel_Date КАК Vyp_k_Sel_Date
		|ИЗ
		|	ВнешнийИсточникДанных.КрымMySQL.Таблица.bid КАК bid
		|		ЛЕВОЕ СОЕДИНЕНИЕ ВнешнийИсточникДанных.КрымMySQL.Таблица.cot КАК cot
		|		ПО bid.BidID = cot.CotID
		|		ЛЕВОЕ СОЕДИНЕНИЕ ВнешнийИсточникДанных.КрымMySQL.Таблица.evt КАК evt
		|		ПО bid.BidID = evt.EvtID
		|ГДЕ
		|	cot.RealAct_NN <> &RealAct_NN
		|	И cot.RealAct_NN ЕСТЬ НЕ NULL 
		|	И (cot.RealAct_Date = &RealAct_Date
		|			ИЛИ cot.RealAct_Date ЕСТЬ NULL)
		|
		|УПОРЯДОЧИТЬ ПО
		|	BidID";
	Запрос.УстановитьПараметр("RealAct_NN", "");
	Запрос.УстановитьПараметр("RealAct_Date", Дата(1,1,1));
	РезультатЗапроса = Запрос.Выполнить();
	
	Выборка = РезультатЗапроса.Выбрать();
	
	Пока Выборка.Следующий() Цикл
		ЗаявкаТП = Документы.ЭЭ_ЗаявкаТП.НайтиПоРеквизиту("ИБР_Основание",Выборка.BidID);
		Если ЗаявкаТП.Ссылка.Пустая() Тогда
			Сообщить("отсуствует в базе : " + Выборка.BidID);
			Продолжить;
			//ЗаявкаОб = Документы.ЭЭ_ЗаявкаТП.СоздатьДокумент();                                          
		иначе
			ЗаявкаОб = ЗаявкаТП.Ссылка.ПолучитьОбъект();
			Если ЗначениеЗаполнено(Выборка.SocTarif) И Выборка.SocTarif > 0  тогда
				ЗаявкаОб.ЛьготнаяКатегория = ЛьготнаяКатегория_;
			Иначе
				ЗаявкаОб.ЛьготнаяКатегория = Неопределено;
			КонецЕсли;
		КонецЕсли;
		
		Если ЗначениеЗаполнено(ЗаявкаОб.ДанныеИсполненияЗаявки) И ЗначениеЗаполнено(Выборка.Vyp_k_Sel_Date) Тогда
			ДанныеИсполненияЗаявки_Об = ЗаявкаОб.ДанныеИсполненияЗаявки.ПолучитьОбъект();
			ДанныеИсполненияЗаявки_Об.ВыполненоКСелектору = Дата(СтрЗаменить(Выборка.Vyp_k_Sel_Date,"-",""));
			ДанныеИсполненияЗаявки_Об.ОбменДанными.Загрузка = Истина;
			ДанныеИсполненияЗаявки_Об.Записать();
		КонецЕсли;
		
		ДатаНачала_ = ?(ЗначениеЗаполнено(Выборка.Dogovor_Date), Выборка.Dogovor_Date,  Дата(1,1,1));
		ДатаОкончания_ = Дата(1,1,1);
		
		Если ЗначениеЗаполнено(ДатаНачала_) И ЗначениеЗаполнено(Выборка.EvtSrok) Тогда 
			ДатаОкончания_ = ДатаНачала_ + (Выборка.EvtSrok * 3600 * 24);
		ИначеЕсли ЗначениеЗаполнено(ДатаНачала_) И ЗначениеЗаполнено(Выборка.EvtSrok) = Ложь Тогда
			ДатаОкончания_ = Дата(1,1,1);
		КонецЕсли;
		
		Если ЗначениеЗаполнено(Выборка.Dogovor_NN) или ЗначениеЗаполнено(Выборка.Dogovor_Date) Тогда 
			Если ЗначениеЗаполнено(ЗаявкаТП.ДоговорТП) Тогда
				Договор_об = ЗаявкаТП.ДоговорТП.ПолучитьОбъект();
			Иначе
				Договор_об = Справочники.ИБР_Договоры.СоздатьЭлемент();
			КонецЕсли;
			
			Договор_об.ДатаЗаключения = ДатаНачала_;    
			Договор_об.ДатаНачала = ДатаНачала_;
			Договор_об.ДатаОкончания = ДатаОкончания_; 
			
			Договор_об.ВидДоговора = УТП_Сервер.ВернутьВидДоговораТП_поЗаявке(ЗаявкаОб.Ссылка);
			Договор_об.ДокументОснование = ЗаявкаОб.Ссылка;  
			Договор_об.Наименование =" Загружен по " + ЗаявкаОб.Ссылка;
			Договор_об.Валюта = Валюта_;
			Договор_об.Организация = ОснКомпания;
			
			Если ЗначениеЗаполнено(Выборка.Dogovor_NN) Тогда
				Договор_об.Код = СокрЛП(Выборка.Dogovor_NN);
			КонецЕсли;
			
			Если ТипЗнч(ЗаявкаОб.Заявитель) = Тип("СправочникСсылка.Контрагенты") Тогда
				Договор_об.ОсновнойКонтрагент = ЗаявкаОб.Заявитель; //может быть строка
			КонецЕсли;
			
			Если ЗначениеЗаполнено(Выборка.CostTotal) Тогда
				Договор_об.ОбщаяСумма = Число(Выборка.CostTotal);
			КонецЕсли;
			
			Договор_об.ОбменДанными.Загрузка = Истина;
			
			Попытка
				Договор_об.Записать();
				ЗаявкаОб.ДоговорТП = Договор_об.Ссылка;
			Исключение
				
			КонецПопытки;

		ИначеЕсли ЗначениеЗаполнено(ЗаявкаТП.ДоговорТП) Тогда
			Дог_об = ЗаявкаТП.ДоговорТП.ПолучитьОбъект();
			Дог_об.Удалить();
			ЗаявкаОб.ДоговорТП = Неопределено;
			
		КонецЕсли;
		
		Если ЗначениеЗаполнено(Выборка.RealAct_NN) или ЗначениеЗаполнено(Выборка.RealAct_Date) Тогда
			
			Если ЗначениеЗаполнено(ЗаявкаТП.АктОТехнологическомПрисоединении) Тогда
				ДокументАКТП_об = ЗаявкаТП.АктОТехнологическомПрисоединении.ПолучитьОбъект();
			Иначе
				ДокументАКТП_об = Документы.ЭЭ_АктОТехнологическомПрисоединении.СоздатьДокумент();
			КонецЕсли;
			
			Попытка      
				Дата_ = Дата(СтрЗаменить(Выборка.RealAct_Date,"-",""));   
			Исключение   
				Дата_ = ДатаОкончания_;//Дата(1,1,1);
			КонецПопытки;
			
			Попытка
				ДокументАКТП_об.Номер = Выборка.RealAct_NN;
			Исключение
				
			КонецПопытки;
			
			ДокументАКТП_об.Дата = Дата_;
			ДокументАКТП_об.Проведен = Истина;
			Попытка
				ДокументАКТП_об.ОбменДанными.Загрузка = Истина;
				ДокументАКТП_об.Записать(); 
				ЗаявкаОб.АктОТехнологическомПрисоединении =  ДокументАКТП_об.Ссылка;
			Исключение
				
			КонецПопытки;

		ИначеЕсли ЗначениеЗаполнено(ЗаявкаТП.АктОТехнологическомПрисоединении) Тогда
			ДокументАКТП_об = ЗаявкаТП.АктОТехнологическомПрисоединении.ПолучитьОбъект();
			ДокументАКТП_об.Удалить();
			ЗаявкаОб.АктОТехнологическомПрисоединении = Неопределено;
		КонецЕсли;
		
		ЗаявкаОб.ОбменДанными.Загрузка = Истина;
		ЗаявкаОб.Записать();
		
	КонецЦикла;
КонецПроцедуры
